7. Modelo físico (SQL)

CREATE TABLE times (
  id SERIAL PRIMARY KEY,
  nome VARCHAR(80) NOT NULL UNIQUE,
  sigla VARCHAR(10) NOT NULL UNIQUE
);

CREATE TABLE usuarios (
  id SERIAL PRIMARY KEY,
  nome VARCHAR(80) NOT NULL,
  email VARCHAR(120) NOT NULL UNIQUE,
  senha_hash VARCHAR(120) NOT NULL,
  time_id INTEGER NOT NULL REFERENCES times(id),
  papel VARCHAR(20) NOT NULL DEFAULT 'torcedor',
  criado_em TIMESTAMP NOT NULL DEFAULT NOW()
);

CREATE TABLE posts (
  id SERIAL PRIMARY KEY,
  usuario_id INTEGER NOT NULL REFERENCES usuarios(id),
  time_id INTEGER NOT NULL REFERENCES times(id),
  texto VARCHAR(280) NOT NULL,
  imagem_url VARCHAR(255),
  criado_em TIMESTAMP NOT NULL DEFAULT NOW()
);

CREATE TABLE comentarios (
  id SERIAL PRIMARY KEY,
  post_id INTEGER NOT NULL REFERENCES posts(id),
  usuario_id INTEGER NOT NULL REFERENCES usuarios(id),
  texto VARCHAR(280) NOT NULL,
  criado_em TIMESTAMP NOT NULL DEFAULT NOW()
);

CREATE TABLE curtidas (
  usuario_id INTEGER NOT NULL REFERENCES usuarios(id),
  post_id INTEGER NOT NULL REFERENCES posts(id),
  PRIMARY KEY (usuario_id, post_id)
);

CREATE TABLE jogos (
  id SERIAL PRIMARY KEY,
  mandante_id INTEGER NOT NULL REFERENCES times(id),
  visitante_id INTEGER NOT NULL REFERENCES times(id),
  data DATE NOT NULL,
  horario VARCHAR(5) NOT NULL,
  placar_mandante INTEGER,
  placar_visitante INTEGER
);
